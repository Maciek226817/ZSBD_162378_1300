CREATE TABLE archiwum_departamentów (
    id NUMBER PRIMARY KEY,
    nazwa VARCHAR2(100),
    data_zamknięcia DATE,
    ostatni_manager VARCHAR2(100)
);


CREATE OR REPLACE TRIGGER after_department_delete
AFTER DELETE ON departments
FOR EACH ROW
BEGIN
    INSERT INTO archiwum_departamentów (id, nazwa, data_zamknięcia, ostatni_manager)
    VALUES (
        :OLD.department_id, 
        :OLD.department_name, 
        SYSDATE, 
        (SELECT first_name || ' ' || last_name 
         FROM employees 
         WHERE employee_id = :OLD.manager_id)
    );
END;
/

CREATE OR REPLACE TRIGGER check_salary_range
BEFORE INSERT OR UPDATE ON employees
FOR EACH ROW
DECLARE
    v_salary NUMBER;
BEGIN
    IF INSERTING THEN
        v_salary := :NEW.salary;
    ELSIF UPDATING THEN
        v_salary := :NEW.salary;
    END IF;

    IF v_salary < 2000 OR v_salary > 26000 THEN
        INSERT INTO złodziej (user_name, czas_zmiany)
        VALUES (USER, SYSDATE);
        RAISE_APPLICATION_ERROR(-20001, 'Zarobki muszą mieścić się w przedziale 2000 - 26000');
    END IF;
END;
/

CREATE SEQUENCE employee_seq
START WITH 1      
INCREMENT BY 1     
NOCACHE           
NOCYCLE;           


CREATE OR REPLACE TRIGGER employee_auto_increment
BEFORE INSERT ON employees
FOR EACH ROW
BEGIN
    IF :NEW.employee_id IS NULL THEN
        SELECT employee_seq.NEXTVAL
        INTO :NEW.employee_id
        FROM dual;
    END IF;
END;
/

CREATE OR REPLACE TRIGGER prevent_salary_update
BEFORE UPDATE ON jobs
FOR EACH ROW
BEGIN
    IF :NEW.min_salary != :OLD.min_salary THEN
        :NEW.min_salary := :OLD.min_salary;
    END IF;
    IF :NEW.max_salary != :OLD.max_salary THEN
        :NEW.max_salary := :OLD.max_salary;
    END IF;
END;
/
