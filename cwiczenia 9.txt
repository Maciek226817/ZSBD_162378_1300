CREATE OR REPLACE PACKAGE regions_pkg AS
    -- CRUD procedury
    PROCEDURE CREATE_REGION(p_region_id IN NUMBER, p_region_name IN VARCHAR2);
    PROCEDURE UPDATE_REGION(p_region_id IN NUMBER, p_region_name IN VARCHAR2);
    PROCEDURE DELETE_REGION(p_region_id IN NUMBER);

    -- Funkcje/procedury do odczytu
    FUNCTION GET_REGION_BY_ID(p_region_id IN NUMBER) RETURN VARCHAR2;
    FUNCTION GET_ALL_REGIONS RETURN SYS_REFCURSOR;
    FUNCTION SEARCH_REGIONS_BY_NAME(p_pattern IN VARCHAR2) RETURN SYS_REFCURSOR;
END regions_pkg;
/


CREATE OR REPLACE PACKAGE BODY regions_pkg AS
    -- Procedura CREATE
    PROCEDURE CREATE_REGION(p_region_id IN NUMBER, p_region_name IN VARCHAR2) IS
    BEGIN
        INSERT INTO REGIONS (REGION_ID, REGION_NAME)
        VALUES (p_region_id, p_region_name);
        COMMIT;
    END CREATE_REGION;

    -- Procedura UPDATE
    PROCEDURE UPDATE_REGION(p_region_id IN NUMBER, p_region_name IN VARCHAR2) IS
    BEGIN
        UPDATE REGIONS
        SET REGION_NAME = p_region_name
        WHERE REGION_ID = p_region_id;
        COMMIT;
    END UPDATE_REGION;

    -- Procedura DELETE
    PROCEDURE DELETE_REGION(p_region_id IN NUMBER) IS
    BEGIN
        DELETE FROM REGIONS
        WHERE REGION_ID = p_region_id;
        COMMIT;
    END DELETE_REGION;

    -- Funkcja GET_REGION_BY_ID
    FUNCTION GET_REGION_BY_ID(p_region_id IN NUMBER) RETURN VARCHAR2 IS
        v_region_name VARCHAR2(25);
    BEGIN
        SELECT REGION_NAME
        INTO v_region_name
        FROM REGIONS
        WHERE REGION_ID = p_region_id;

        RETURN v_region_name;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RETURN 'Region not found';
    END GET_REGION_BY_ID;

    -- Funkcja GET_ALL_REGIONS
    FUNCTION GET_ALL_REGIONS RETURN SYS_REFCURSOR IS
        v_cursor SYS_REFCURSOR;
    BEGIN
        OPEN v_cursor FOR
            SELECT REGION_ID, REGION_NAME
            FROM REGIONS;
        RETURN v_cursor;
    END GET_ALL_REGIONS;

    -- Funkcja SEARCH_REGIONS_BY_NAME
    FUNCTION SEARCH_REGIONS_BY_NAME(p_pattern IN VARCHAR2) RETURN SYS_REFCURSOR IS
        v_cursor SYS_REFCURSOR;
    BEGIN
        OPEN v_cursor FOR
            SELECT REGION_ID, REGION_NAME
            FROM REGIONS
            WHERE REGION_NAME LIKE '%' || p_pattern || '%';
        RETURN v_cursor;
    END SEARCH_REGIONS_BY_NAME;

END regions_pkg;

CREATE TABLE AUDIT_LOG (
    LOG_ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    LOG_DATE TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ERROR_CODE NUMBER,
    ERROR_MESSAGE VARCHAR2(4000),
    MODULE_NAME VARCHAR2(50),
    OPERATION_NAME VARCHAR2(50)
);


CREATE OR REPLACE PACKAGE regions_pkg AS
    -- Niestandardowe wyjątki
    e_duplicate_region_name EXCEPTION;
    e_region_with_countries EXCEPTION;

    -- Procedury CRUD
    PROCEDURE CREATE_REGION(p_region_id IN NUMBER, p_region_name IN VARCHAR2);
    PROCEDURE UPDATE_REGION(p_region_id IN NUMBER, p_region_name IN VARCHAR2);
    PROCEDURE DELETE_REGION(p_region_id IN NUMBER);

    -- Funkcje odczytu
    FUNCTION GET_REGION_BY_ID(p_region_id IN NUMBER) RETURN VARCHAR2;
    FUNCTION GET_ALL_REGIONS RETURN SYS_REFCURSOR;
    FUNCTION SEARCH_REGIONS_BY_NAME(p_pattern IN VARCHAR2) RETURN SYS_REFCURSOR;

    -- Procedura logująca błędy
    PROCEDURE LOG_ERROR(
        p_error_code IN NUMBER,
        p_error_message IN VARCHAR2,
        p_module_name IN VARCHAR2,
        p_operation_name IN VARCHAR2
    );
END regions_pkg;
/


CREATE OR REPLACE PACKAGE BODY regions_pkg AS
    -- Niestandardowe wyjątki
    e_duplicate_region_name EXCEPTION; -- Deklaracja tylko raz
    e_region_with_countries EXCEPTION; -- Deklaracja tylko raz

    -- Zwracanie kodów błędów dla wyjątków
    PRAGMA EXCEPTION_INIT(e_duplicate_region_name, -20001);
    PRAGMA EXCEPTION_INIT(e_region_with_countries, -20002);

    -- Procedura logująca błędy
    PROCEDURE LOG_ERROR(
        p_error_code IN NUMBER,
        p_error_message IN VARCHAR2,
        p_module_name IN VARCHAR2,
        p_operation_name IN VARCHAR2
    ) IS
    BEGIN
        INSERT INTO AUDIT_LOG (ERROR_CODE, ERROR_MESSAGE, MODULE_NAME, OPERATION_NAME)
        VALUES (p_error_code, p_error_message, p_module_name, p_operation_name);
        COMMIT;
    END LOG_ERROR;

    -- Procedura CREATE_REGION
    PROCEDURE CREATE_REGION(p_region_id IN NUMBER, p_region_name IN VARCHAR2) IS
        v_count NUMBER;
    BEGIN
        -- Sprawdzanie czy region o podanej nazwie już istnieje
        SELECT COUNT(*)
        INTO v_count
        FROM REGIONS
        WHERE UPPER(REGION_NAME) = UPPER(p_region_name);

        IF v_count > 0 THEN
            RAISE e_duplicate_region_name;
        END IF;

        -- Dodanie regionu
        INSERT INTO REGIONS (REGION_ID, REGION_NAME)
        VALUES (p_region_id, p_region_name);
        COMMIT;
    EXCEPTION
        WHEN e_duplicate_region_name THEN
            LOG_ERROR(-20001, 'Region name already exists: ' || p_region_name, 'REGIONS', 'CREATE_REGION');
            RAISE_APPLICATION_ERROR(-20001, 'Region with this name already exists.');
        WHEN OTHERS THEN
            LOG_ERROR(SQLCODE, SQLERRM, 'REGIONS', 'CREATE_REGION');
            RAISE;
    END CREATE_REGION;

    -- Procedura DELETE_REGION
    PROCEDURE DELETE_REGION(p_region_id IN NUMBER) IS
        v_count NUMBER;
    BEGIN
        -- Sprawdzanie czy region ma przypisane kraje
        SELECT COUNT(*)
        INTO v_count
        FROM COUNTRIES -- Zakładamy, że istnieje tabela COUNTRIES
        WHERE REGION_ID = p_region_id;

        IF v_count > 0 THEN
            RAISE e_region_with_countries;
        END IF;

        -- Usuwanie regionu
        DELETE FROM REGIONS
        WHERE REGION_ID = p_region_id;
        COMMIT;
    EXCEPTION
        WHEN e_region_with_countries THEN
            LOG_ERROR(-20002, 'Cannot delete region with assigned countries: ' || p_region_id, 'REGIONS', 'DELETE_REGION');
            RAISE_APPLICATION_ERROR(-20002, 'Cannot delete region with assigned countries.');
        WHEN OTHERS THEN
            LOG_ERROR(SQLCODE, SQLERRM, 'REGIONS', 'DELETE_REGION');
            RAISE;
    END DELETE_REGION;

    -- Procedury UPDATE i funkcje GET pozostają bez zmian
    PROCEDURE UPDATE_REGION(p_region_id IN NUMBER, p_region_name IN VARCHAR2) IS
    BEGIN
        UPDATE REGIONS
        SET REGION_NAME = p_region_name
        WHERE REGION_ID = p_region_id;
        COMMIT;
    EXCEPTION
        WHEN OTHERS THEN
            LOG_ERROR(SQLCODE, SQLERRM, 'REGIONS', 'UPDATE_REGION');
            RAISE;
    END UPDATE_REGION;

    FUNCTION GET_REGION_BY_ID(p_region_id IN NUMBER) RETURN VARCHAR2 IS
        v_region_name VARCHAR2(25);
    BEGIN
        SELECT REGION_NAME
        INTO v_region_name
        FROM REGIONS
        WHERE REGION_ID = p_region_id;

        RETURN v_region_name;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            LOG_ERROR(-1, 'Region not found for ID: ' || p_region_id, 'REGIONS', 'GET_REGION_BY_ID');
            RETURN 'Region not found';
        WHEN OTHERS THEN
            LOG_ERROR(SQLCODE, SQLERRM, 'REGIONS', 'GET_REGION_BY_ID');
            RAISE;
    END GET_REGION_BY_ID;

    FUNCTION GET_ALL_REGIONS RETURN SYS_REFCURSOR IS
        v_cursor SYS_REFCURSOR;
    BEGIN
        OPEN v_cursor FOR
            SELECT REGION_ID, REGION_NAME
            FROM REGIONS;
        RETURN v_cursor;
    END GET_ALL_REGIONS;

    FUNCTION SEARCH_REGIONS_BY_NAME(p_pattern IN VARCHAR2) RETURN SYS_REFCURSOR IS
        v_cursor SYS_REFCURSOR;
    BEGIN
        OPEN v_cursor FOR
            SELECT REGION_ID, REGION_NAME
            FROM REGIONS
            WHERE REGION_NAME LIKE '%' || p_pattern || '%';
        RETURN v_cursor;
    END SEARCH_REGIONS_BY_NAME;

END regions_pkg;
